1. git clone - клонирует репозиторий
2. git status - показывает состояние
3. git add - добавляет файлы
4. git commit - создаёт коммит
5. git push - отправляет на сервер
6. git pull - забирает изменения с удалённого репозитория
7. git branch - показывает список веток или создаёт новую
8. git checkout - переключается между ветками
9. git merge - сливает изменения из одной ветки в другую
10. git log - показывает историю коммитов
